const NAVY = '#16326b', SOFTBLUE = '#e6efff', SOFTGREEN = '#e7f6ec', SOFTORANGE = '#fff0e0';

function titleCard(c, T, a = 1) { at(c, W / 2, 250, 1, c => {

function pchip(c, x, y, t, s, bg = '#fff', col = INK, size = 46, font = 'Noto') { if (s <= 0) return; at(c, x, y, s, c => {
  c.font = `bold ${size}px ${font}`; const w = c.measureText(t).width + 64, h = size * 1.75;
  card(c, -w / 2, -h / 2, w, h, h / 2, bg, 5); text(c, t, 0, size * .36, size, col, {font});
}); }

function signpost(c, x, y, T) {
  line(c, [[x, y], [x, y - 620]], 10);
  const arr = [['ERP?', BLUE, .25], ['Aplikasi?', GREEN, .6], ['Website?', PURPLE, .95]];
  arr.forEach(([t, col, a], i) => { const s = pop(T, a), wob = T > 2.8 ? Math.sin(T * 7 + i * 2) * .07 : 0;

}

// progress tracker for the 5 steps
const STEPS = ['Chat', 'Konsultasi', 'Desain', 'Bayar', 'Jadi'];
function tracker(c, y, cur, k = 1, T = 0) { // cur: 0..4 active index (float allowed for fill)
  const x0 = 140, x1 = 940, n = STEPS.length;
  line(c, [[x0, y], [x1, y]], 10, '#d9cdb6');
  const fx = lerp(x0, x1, clamp(cur / (n - 1)));
  if (cur > 0) line(c, [[x0, y], [fx, y]], 10, ORANGE);
  STEPS.forEach((t, i) => { const x = lerp(x0, x1, i / (n - 1)), s = clamp(k * n - i);
    if (s <= 0) return;
    const done = i < Math.floor(cur + .001), act = i === Math.round(cur);
    at(c, x, y, back(clamp(s)), c => {
      circ(c, 0, 0, act ? 44 : 34, done ? GREEN : act ? ORANGE : '#fff', 6);
      if (done) check(c, 0, 2, 16, 8, '#fff'); else text(c, String(i + 1), 0, act ? 20 : 16, act ? 60 : 46, act ? '#fff' : MUTED);
      text(c, t, 0, 92, act ? 46 : 38, act ? INK : MUTED, {font: 'Noto'});
    });
  });
}
function stepHead(c, n, title, sub, T, a) {
  at(c, 0, 0, 1, c => {
    const s = pop(T, a); at(c, 150, 470, s, c => { circ(c, 0, 0, 62, ORANGE, 7); text(c, String(n), 0, 30, 92, '#fff'); });
    writeOn(c, title, 600, 498, 100, INK, P(T, a + .1, a + .6), {});
    if (sub) { const k = pe(T, a + .45, .4); at(c, 600, 572, 1, c => text(c, sub, 0, 0, 42, MUTED, {font: 'Noto', bold: false}), k); }
  });
}
