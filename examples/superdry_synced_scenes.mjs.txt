function sdStore(c, x, y, w, h, sign, signCol = BLUE) { rect(c, x, y - h, w, h, '#fff7e8', 6); rect(c, x - 14, y - h - 70, w + 28, 80, signCol, 6); text(c, sign, x + w / 2, y - h - 14, 50, '#fff');

function sdTote(c, x, y, s = 1) { at(c, x, y, s, c => { c.beginPath(); c.arc(0, -70, 45, Math.PI, 0); c.lineWidth = 9; c.strokeStyle = INK; c.stroke(); rect(c, -95, -70, 190, 200, ORANGE, 6); text(c, 'EDISI', 0, 0, 40, '#fff'); text(c, 'TERBATAS', 0, 46, 36, '#fff'); }); }
